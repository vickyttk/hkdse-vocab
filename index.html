    </div>


    <div id="app" style="display:none;">

        <div class="progress-container"><div id="progress" class="progress-bar"></div></div>

        <p id="stats">Words Mastered: 0/10</p>


        <div class="card">

            <h2 id="wordDisplay">Loading...</h2>

            <button onclick="speak()">🔊 Listen</button>

            <p id="defDisplay"></p>

            <input type="text" id="answerInput" placeholder="Type spelling here..." style="display:none;">

        </div>


        <button id="mainBtn" onclick="nextQuestion()">Next Word</button>

        <button id="submitBtn" onclick="submitProgress()" style="display:none; background:#34C759;">Submit Results to Teacher</button>

    <script>

        const vocab = [

            {w: "alleviate", d: "to make suffering less severe"},

            {w: "detrimental", d: "tending to cause harm"},

            {w: "ubiquitous", d: "found everywhere"},

            {w: "pragmatic", d: "sensible and realistic"}

        ];


        let score = 0;

        let currentWord = {};

        let goal = 10;


        function startApp() {

            if(!document.getElementById('studentName').value) return alert("Enter name!");

            document.getElementById('setup').style.display = 'none';

            document.getElementById('app').style.display = 'block';

            nextQuestion();

        }


        function speak() {

            let msg = new SpeechSynthesisUtterance(currentWord.w);

            msg.lang = 'en-GB';

            window.speechSynthesis.speak(msg);

        }


        function nextQuestion() {

            currentWord = vocab[Math.floor(Math.random() * vocab.length)];

            document.getElementById('wordDisplay').innerText = "???";

            document.getElementById('defDisplay').innerText = currentWord.d;

            document.getElementById('answerInput').style.display = 'block';

            document.getElementById('answerInput').value = "";

            document.getElementById('mainBtn').onclick = checkAnswer;

            document.getElementById('mainBtn').innerText = "Check Spelling";

        }


        function checkAnswer() {

            let userAns = document.getElementById('answerInput').value.trim().toLowerCase();

            if(userAns === currentWord.w.toLowerCase()) {

                score++;

                alert("Correct!");

            } else {

                alert("Incorrect. It was: " + currentWord.w);

            }

            updateUI();

        }


        function updateUI() {

            let percent = (score / goal) * 100;

            document.getElementById('progress').style.width = percent + "%";

            document.getElementById('stats').innerText = `Words Mastered: ${score}/${goal}`;

            

            if(score >= goal) {

                document.getElementById('submitBtn').style.display = 'inline-block';

            }

            nextQuestion();

        }


        function submitProgress() {

            const name = document.getElementById('studentName').value;

            // For a real teacher dashboard, we send this to a Google Form URL

            const formURL = "YOUR_GOOGLE_FORM_ENDPOINT_HERE";

            alert(`Results for ${name} submitted! Score: ${score}`);

            // You can use fetch(formURL) here to actually save the data

        }
